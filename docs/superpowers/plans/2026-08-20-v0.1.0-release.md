> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `v0.1.0` with native MiniZinc set/enum values, retained diagnostics, documented MIT licensing, and automated compatibility checks.

**Architecture:** Keep the public solve surface unchanged. Add standard-library JSON adapters for MiniZinc-specific values, collect warning/checker messages in the existing three solve loops through one small internal accumulator, and validate the same repository on pinned GitHub Actions and official MiniZinc bundles.

**Tech Stack:** Go 1.21+, `encoding/json`, MiniZinc JSON stream 2.6+, GitHub Actions, golangci-lint 2.13.1, GitHub CLI.

func TestSetJSON(t *testing.T) {
	value := Set[int]{
		Elements: []int{1, 8},
		Ranges:   []SetRange[int]{{Min: 3, Max: 5}},
	}
	data, err := json.Marshal(value)
	if err != nil {
		t.Fatal(err)
	}
	if string(data) != `{"set":[1,8,[3,5]]}` {
		t.Fatalf("got %s", data)
	}

	var decoded Set[int]
	if err := json.Unmarshal([]byte(`{"set":[1,[3,5],8]}`), &decoded); err != nil {
		t.Fatal(err)
	}
	if !reflect.DeepEqual(decoded.Elements, []int{1, 8}) ||
		!reflect.DeepEqual(decoded.Ranges, []SetRange[int]{{Min: 3, Max: 5}}) {
		t.Fatalf("got %+v", decoded)
	}
}

func TestEnumJSON(t *testing.T) {
	tests := []struct {
		name  string
		value Enum
		json  string
	}{
		{name: "value", value: Enum{Value: "Canada"}, json: `{"e":"Canada"}`},
		{name: "nested", value: Enum{Constructor: "C", Argument: Enum{Value: "Canada"}}, json: `{"c":"C","e":{"e":"Canada"}}`},
		{name: "integer", value: Enum{Constructor: "Right", Argument: 2}, json: `{"c":"Right","e":2}`},
	}
	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			data, err := json.Marshal(test.value)
			if err != nil {
				t.Fatal(err)
			}
			if string(data) != test.json {

				t.Fatal(err)
			}
			data, err = json.Marshal(decoded)
			if err != nil {
				t.Fatal(err)
			}
			if string(data) != test.json {
				t.Fatalf("round trip got %s", data)
			}
		})
	}
}

Create `values.go` with these public types and methods:

```go
package minizinc

import (
	"bytes"
	"encoding/json"
	"io"
)

type SetRange[T any] struct {
	Min T
	Max T
}

type Set[T any] struct {
	Elements []T
	Ranges   []SetRange[T]
}

func (s Set[T]) MarshalJSON() ([]byte, error) {
	items := make([]any, 0, len(s.Elements)+len(s.Ranges))
	for _, element := range s.Elements {
		items = append(items, element)
	}
	for _, itemRange := range s.Ranges {
		items = append(items, [2]T{itemRange.Min, itemRange.Max})
	}
	return json.Marshal(struct {
		Set []any `json:"set"`
	}{Set: items})
}

func (s *Set[T]) UnmarshalJSON(data []byte) error {
	if s == nil {
		return newError("cannot unmarshal set into nil receiver")
	}
	var object map[string]json.RawMessage
	if err := json.Unmarshal(data, &object); err != nil {
		return wrapError("invalid MiniZinc set", err)
	}
	raw, ok := object["set"]
	if !ok || len(object) != 1 {
		return newError("invalid MiniZinc set")
	}
	var items []json.RawMessage
	if err := json.Unmarshal(raw, &items); err != nil {
		return wrapError("invalid MiniZinc set", err)
	}
	elements := make([]T, 0, len(items))
	ranges := make([]SetRange[T], 0)
	for _, item := range items {
		trimmed := bytes.TrimSpace(item)
		if len(trimmed) > 0 && trimmed[0] == '[' {
			var bounds []json.RawMessage
			if err := json.Unmarshal(item, &bounds); err != nil || len(bounds) != 2 {
				return newError("invalid MiniZinc set range")
			}
			var itemRange SetRange[T]
			if err := json.Unmarshal(bounds[0], &itemRange.Min); err != nil {
				return wrapError("invalid MiniZinc set range minimum", err)
			}
			if err := json.Unmarshal(bounds[1], &itemRange.Max); err != nil {
				return wrapError("invalid MiniZinc set range maximum", err)
			}
			ranges = append(ranges, itemRange)
			continue
		}
		var element T
		if err := json.Unmarshal(item, &element); err != nil {
			return wrapError("invalid MiniZinc set element", err)
		}
		elements = append(elements, element)
	}
	s.Elements = elements
	s.Ranges = ranges
	return nil
}

type Enum struct {
	Value       string
	Constructor string
	Argument    any
}

func (e Enum) MarshalJSON() ([]byte, error) {
	if e.Value != "" && e.Constructor == "" && e.Argument == nil {
		return json.Marshal(struct {
			Value string `json:"e"`
		}{Value: e.Value})
	}
	if e.Value == "" && e.Constructor != "" && e.Argument != nil {
		argument, err := json.Marshal(e.Argument)
		if err != nil {
			return nil, wrapError("invalid MiniZinc enum constructor argument", err)
		}
		if bytes.Equal(argument, []byte("null")) {
			return nil, newError("invalid MiniZinc enum constructor argument")
		}
		return json.Marshal(struct {
			Constructor string          `json:"c"`
			Argument    json.RawMessage `json:"e"`
		}{Constructor: e.Constructor, Argument: argument})
	}
	return nil, newError("invalid MiniZinc enum")
}

func (e *Enum) UnmarshalJSON(data []byte) error {
	if e == nil {
		return newError("cannot unmarshal enum into nil receiver")
	}
	var object map[string]json.RawMessage
	if err := json.Unmarshal(data, &object); err != nil {
		return wrapError("invalid MiniZinc enum", err)
	}
	argument, hasArgument := object["e"]
	constructor, hasConstructor := object["c"]
	if !hasArgument || len(object) != 1 && !(hasConstructor && len(object) == 2) {
		return newError("invalid MiniZinc enum")
	}
	if !hasConstructor {
		var value string
		if err := json.Unmarshal(argument, &value); err != nil || value == "" {
			return newError("invalid MiniZinc enum value")
		}
		*e = Enum{Value: value}
		return nil
	}
	var name string
	if err := json.Unmarshal(constructor, &name); err != nil || name == "" {
		return newError("invalid MiniZinc enum constructor")
	}
	trimmed := bytes.TrimSpace(argument)
	var value any
	if len(trimmed) > 0 && trimmed[0] == '{' {
		var nested Enum
		if err := json.Unmarshal(argument, &nested); err != nil {
			return err
		}
		value = nested
	} else {
		decoder := json.NewDecoder(bytes.NewReader(argument))
		decoder.UseNumber()
		if err := decoder.Decode(&value); err != nil {
			return wrapError("invalid MiniZinc enum constructor argument", err)
		}
		if value == nil || decoder.Decode(&struct{}{}) != io.EOF {
			return newError("invalid MiniZinc enum constructor argument")
		}
	}
	*e = Enum{Constructor: name, Argument: value}
	return nil
}
```

```go
type solveDiagnostics struct {
	warnings []Warning
	checker  []CheckerMessage
}

func (d *solveDiagnostics) add(message streamMessage) {
	switch message.Type {
	case "warning":
		d.warnings = append(d.warnings, Warning{
			Message:  message.Message,
			Location: deepCopyValue(message.Location),
			Stack:    deepCopyValue(message.Stack).([]any),
		})
	case "checker":
		for _, nested := range message.Messages {
			d.checker = append(d.checker, CheckerMessage{
				Type:         nested.Type,
				Status:       nested.Status,
				Output:       deepCopyValue(nested.Output).(map[string]any),
				SectionOrder: append([]string(nil), nested.Sections...),
				Statistics:   deepCopyValue(nested.Statistics).(map[string]any),
				What:         nested.What,
				Message:      nested.Message,
				Location:     deepCopyValue(nested.Location),
				Stack:        deepCopyValue(nested.Stack).([]any),
			})
		}
	}
}

func (d *solveDiagnostics) attach(result *Result) {
	result.Warnings = append([]Warning(nil), d.warnings...)
	result.CheckerMessages = append([]CheckerMessage(nil), d.checker...)
}

func (d *solveDiagnostics) hasMessages() bool {
	return len(d.warnings) > 0 || len(d.checker) > 0
}
```

Create `.github/workflows/ci.yml` with `contents: read`, concurrency

```yaml
name: CI

on:
  push:
    branches: [main]
  pull_request:
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  test:
    name: Test Go ${{ matrix.go }}
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        go: ['1.21.x', stable]
    steps:
      - uses: actions/checkout@v7.0.1
      - uses: actions/setup-go@v7.0.0
        with:
          go-version: ${{ matrix.go }}
          cache: false
      - run: go test -count=1 ./...
      - run: go vet ./...

  race:
    name: Race
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7.0.1
      - uses: actions/setup-go@v7.0.0
        with:
          go-version: stable
          cache: false
      - run: go test -race -count=1 ./...

  lint:
    name: Lint
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7.0.1
      - uses: actions/setup-go@v7.0.0
        with:
          go-version: stable
          cache: false
      - uses: golangci/golangci-lint-action@v9.3.0
        with:
          version: v2.13.1

  cross-build:
    name: Build ${{ matrix.goos }}/${{ matrix.goarch }}
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        include:
          - goos: linux
            goarch: amd64
          - goos: windows
            goarch: amd64
          - goos: darwin
            goarch: arm64
    steps:
      - uses: actions/checkout@v7.0.1
      - uses: actions/setup-go@v7.0.0
        with:
          go-version: stable
          cache: false
      - run: go build ./...
        env:
          CGO_ENABLED: '0'
          GOOS: ${{ matrix.goos }}
          GOARCH: ${{ matrix.goarch }}

  integration:
    name: MiniZinc ${{ matrix.version }}
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        include:
          - version: 2.6.4
            sha256: 2e3c2ad1f40c4d45e3af7b24f5445c520c895cdb1d6f390e1b3d679968933414
          - version: 2.10.0
            sha256: 1781daef871a52bebc992b21049ed514cfcdf7cc652ce26df8c9fa82aca1a852
    steps:
      - uses: actions/checkout@v7.0.1
      - uses: actions/setup-go@v7.0.0
        with:
          go-version: stable
          cache: false
      - name: Install MiniZinc
        env:
          MINIZINC_VERSION: ${{ matrix.version }}
          MINIZINC_SHA256: ${{ matrix.sha256 }}
        run: |
          archive="MiniZincIDE-${MINIZINC_VERSION}-bundle-linux-x86_64.tgz"
          curl -fsSLO "https://github.com/MiniZinc/MiniZincIDE/releases/download/${MINIZINC_VERSION}/${archive}"
          echo "${MINIZINC_SHA256}  ${archive}" | sha256sum -c -
          mkdir "$RUNNER_TEMP/minizinc"
          tar -xzf "$archive" --strip-components=1 -C "$RUNNER_TEMP/minizinc"
          echo "$RUNNER_TEMP/minizinc/bin" >> "$GITHUB_PATH"
      - name: Verify solvers
        run: |
          minizinc --version
          minizinc --solvers | grep -i gecode
          minizinc --solvers | grep -i coin-bc
      - name: Run required integrations
        run: |
          output="$(go test -count=1 -v -run 'TestDriver|TestFindSolver|TestSimpleSolve|TestSolveWithParams|TestMiniZincValues|TestSolveAll|TestGecodeSolverOptions|TestCoinBCSolverOptions' .)"
          printf '%s\n' "$output"
          if grep -q -- '--- SKIP' <<<"$output"; then
            exit 1
          fi
      - run: go test -count=1 ./...
```
